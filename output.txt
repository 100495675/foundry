     3	/// Internal infrastructure trait to statically resolve the evaluation
     4	/// of any pure executable logic (Function Items or function pointers) during fallback.
     5	pub trait Pour {
     6	    type Output;
     7	    fn pour(&self) -> Self::Output;
     8	}
     9	
    10	// Universal Blanket Implementation for anything that satisfies `Fn() -> R + Copy`.
    11	// Seamlessly covers standard function pointers and clean Function Items without overlapping.
    12	impl<R, G> Pour for G
    13	where
    14	    G: Fn() -> R + Copy,
    15	{
    16	    type Output = R;
    17	
    18	    #[inline(always)]
    19	    fn pour(&self) -> R {
    20	        (self)()
    21	    }
    22	}
    23	
    24	/// Universal zero-cost runtime wrapper.
    25	///
    26	/// Occupies exactly 24 bytes on the Stack and is as lightweight as a primitive type.
    27	/// The `F` type mathematically maps the unique identity of the function in the foundry.
    28	pub struct Mold<F> {
    29	    fallback_function: F,
    30	    injected_bytes: Option<&'static [u8]>,
    31	}
    32	
    33	impl<F: Clone> Clone for Mold<F> {
    34	    #[inline(always)]
    35	    fn clone(&self) -> Self {
    36	        Self {
    37	            fallback_function: self.fallback_function.clone(),
    38	            injected_bytes: self.injected_bytes,
    39	        }
    40	    }
    41	}

    45	impl<F> Mold<F>
    46	where
    47	    F: Pour + Copy,
    48	    <F as Pour>::Output: serde::de::DeserializeOwned,
    49	{
    50	    /// Internal constructor for zero-allocation on the Heap.
    51	    #[inline(always)]
    52	    #[doc(hidden)]
    53	    pub const fn new_internal(fallback_function: F, injected_bytes: Option<&'static [u8]>) -> Self {
    54	        Self {
    55	            fallback_function,
    56	            injected_bytes,
    57	        }
    58	    }
    59	
    60	    /// Deserializes from the static injected bytes (Forged Phase) or pours the original function (Dynamic Phase).
    61	    #[inline(always)]
    62	    pub fn cast(&self) -> <F as Pour>::Output {
    63	        if let Some(matrix_bytes) = self.injected_bytes {
    64	            match cast_from_matrix::<<F as Pour>::Output>(matrix_bytes) {
    65	                Ok(object) => return object,
    66	                Err(e) => panic!("foundry: cast_from_matrix failed: {:?}", e),
    67	            }
    68	        }
    69	
    70	        // Pours the original function: zero Box, zero vtable, 100% inlinable.
    71	        self.fallback_function.pour()
    72	    }
    73	
    74	    /// Strict binary guarantee: returns `true` only if the physical binary
    75	    /// carries the pre-calculated bytes injected from the `.matrix` file.
    76	    #[inline(always)]
    77	    pub const fn is_forged(&self) -> bool {
    78	        self.injected_bytes.is_some()
    79	    }
    80	}
